Main Module¶

turtle_chessboard ¶

Chess Board Drawer¶

A visual chess board creator and customizer built with Python's Turtle graphics.

This module provides an interactive application for generating chess boards with customizable sizes, color themes, and animation options. It features a clean user interface with buttons for theme selection, board size adjustment, and animation toggling.

Features¶

  • Multiple color themes (Classic, Vintage, Purple, Gray, Blue, Red)
  • Adjustable board sizes (4×4 to 16×16)
  • Animation on/off toggle
  • Integration with a chess game module

Usage¶

# Run the chess board drawer application
python turtle_chessboard.py

Dependencies¶

  • Python 3.6+
  • Turtle graphics (included in standard Python installation)
  • Tkinter (for chess game integration)

def draw_board(size=8, square_size=50, theme_name="Classic"):
    """
    Draw a complete chess board with the given parameters.

    This function creates a chess board centered on the screen with the specified
    size, square dimensions, and color theme. For standard 8×8 boards, it also
    adds row and column labels (a-h, 1-8).

    Args:
        size (int): Number of squares per side (default: 8)
        square_size (int): Size of each square in pixels (default: 50)
        theme_name (str): Name of the color theme to use (default: "Classic")

    Returns:
        None

    Note:
        The board is drawn centered on the screen, with the origin (0,0) at the center.
        Row and column labels are only added for standard 8×8 boards.
    """
    # Get colors from the theme
    light_color = color_themes[theme_name]["light"]
    dark_color = color_themes[theme_name]["dark"]

    # Calculate the total board size
    board_size = size * square_size

    # Calculate starting position to center the board
    start_x = -board_size / 2
    start_y = -board_size / 2 + 100  # Moved up to reduce top padding

    # Clear previous drawings
    board_drawer.clear()

    # Draw all squares
    for row in range(size):
        for col in range(size):
            # Determine square color (alternating pattern)
            color = light_color if (row + col) % 2 == 0 else dark_color

            # Calculate position for this square
            x = start_x + col * square_size
            y = start_y + row * square_size

            # Draw the square
            draw_square(board_drawer, x, y, square_size, color)

    # Draw border around the board
    board_drawer.penup()
    board_drawer.goto(start_x, start_y)
    board_drawer.pendown()
    board_drawer.pensize(1)  # Reduced border size
    board_drawer.color("black")
    for _ in range(4):
        board_drawer.forward(board_size)
        board_drawer.left(90)

    # Draw labels (a-h, 1-8) for a standard 8x8 board
    if size == 8:
        # Draw column labels (a-h)
        board_drawer.penup()
        for col in range(8):
            x = start_x + col * square_size + square_size / 2
            y = start_y - 25
            board_drawer.goto(x, y)
            board_drawer.write(chr(97 + col), align="center", font=("Arial", 12, "normal"))

        # Draw row labels (1-8)
        for row in range(8):
            x = start_x - 25
            y = start_y + row * square_size + square_size / 2 - 6
            board_drawer.goto(x, y)
            board_drawer.write(str(row + 1), align="center", font=("Arial", 12, "normal"))

    screen.update()  # Update the screen after drawing

def draw_button(t, x, y, width, height, fill_color, text, text_color="black"):
    """
    Draw an interactive button with text.

    Args:
        t (turtle.Turtle): The turtle object to use for drawing
        x (float): X-coordinate of the bottom-left corner
        y (float): Y-coordinate of the bottom-left corner
        width (float): Width of the button in pixels
        height (float): Height of the button in pixels
        fill_color (str): Color name or hex code for the button background
        text (str): Text to display on the button
        text_color (str, optional): Color for the button text (default: "black")

    Returns:
        None

    Example:
        ```python
        draw_button(ui_turtle, -100, -250, 80, 30, "lightblue", "Click Me")
        ```
    """
    t.penup()
    t.goto(x, y)
    t.pendown()

    # Draw button background
    t.fillcolor(fill_color)
    t.begin_fill()
    for _ in range(2):
        t.forward(width)
        t.left(90)
        t.forward(height)
        t.left(90)
    t.end_fill()

    # Add text
    t.penup()
    t.goto(x + width/2, y + height/2 - 10)
    t.color(text_color)
    t.write(text, align="center", font=("Arial", 12, "bold"))
    t.color("black")  # Reset color

def draw_ui_elements():
    """
    Draw all UI controls and labels for the application.

    This function creates:
    - Theme selector buttons for each available color theme
    - Board size controls (+ and - buttons)
    - Animation toggle button
    - Play Game button

    Returns:
        None

    Note:
        This function uses the global variables `current_size`, `current_theme`,
        and `animation_on` to determine the current state of the UI elements.
    """
    global current_size, current_theme

    ui_drawer.clear()

    # Define y positions
    theme_y = -250
    size_y = -308
    play_y = -384

    # 1. Draw theme selector label and buttons
    ui_drawer.penup()
    ui_drawer.goto(-290, theme_y + 5)
    ui_drawer.write("Themes: ", align="left", font=("Arial", 14, "bold"))

    # Draw theme buttons
    button_width = 70
    button_height = 30
    x_pos = -200

    for theme in color_themes:
        # Use the theme's dark color for the button
        button_color = color_themes[theme]["dark"]

        # Determine proper text color - white for dark backgrounds
        text_color = "white"

        # Draw the button
        draw_button(ui_drawer, x_pos, theme_y, button_width, button_height, 
                   button_color, theme, text_color)

        # Draw border for all buttons (white/transparent for non-selected)
        ui_drawer.penup()
        ui_drawer.goto(x_pos, theme_y)
        ui_drawer.pendown()

        # Black border for active theme, white for others
        if theme == current_theme:
            ui_drawer.pensize(1)
            ui_drawer.color("black")
        else:
            ui_drawer.pensize(1)
            ui_drawer.color("white")

        for _ in range(2):
            ui_drawer.forward(button_width)
            ui_drawer.left(90)
            ui_drawer.forward(button_height)
            ui_drawer.left(90)
        ui_drawer.pensize(1)
        ui_drawer.color("black")  # Reset color for next operations

        x_pos += button_width + 10

    # 2. Draw board size controls and animation toggle on the same row
    ui_drawer.penup()
    ui_drawer.goto(-290, size_y)
    ui_drawer.write("Board Size:", align="left", font=("Arial", 14, "bold"))

    # Draw decrease size button (minus)
    draw_button(ui_drawer, -170, size_y, 40, 25, "lightgray", "−")

    # Draw current size label - vertically centered between buttons
    ui_drawer.penup()
    ui_drawer.goto(-100, size_y)  # Centered between buttons
    ui_drawer.write(f"{current_size}×{current_size}", align="center", font=("Arial", 14, "bold"))

    # Draw increase size button (plus)
    draw_button(ui_drawer, -70, size_y, 40, 25, "lightgray", "+")

    # Draw Animation Toggle button (on same row as size controls)
    ui_drawer.penup()
    ui_drawer.goto(100, size_y)
    ui_drawer.write("Animation:", align="left", font=("Arial", 14, "bold"))

    # Draw the toggle button
    animation_state = "ON" if animation_on else "OFF"
    animation_color = "lightgreen" if animation_on else "lightgray"
    draw_button(ui_drawer, 210, size_y, 60, 25, animation_color, animation_state)

    # 3. Draw Play Game button - centered and larger
    draw_button(ui_drawer, -150, play_y, 300, 50, "green", "Play Game", "white")

    screen.update()

    message_turtle.hideturtle()
    message_turtle.penup()
    message_turtle.goto(0, 0)
    message_turtle.color("green")
    message_turtle.write("Welcome to Chess Game!\n\nThis is a placeholder for the actual game.\n\nClick anywhere to return to the board drawer.", 
                       align="center", font=("Arial", 20, "bold"))

    # Wait for click to go back
    def go_back(x, y):
        message_turtle.clear()
        # Redraw the board
        square_size = min(600 // current_size, 70)
        draw_board(current_size, square_size, current_theme)
        draw_ui_elements()
        # Reset the click handler
        screen.onclick(handle_click)

    screen.onclick(go_back)
